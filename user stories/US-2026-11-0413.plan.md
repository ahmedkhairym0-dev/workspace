# Technical Implementation Plan: Design Child-Table Excel Import/Export Controls (US-2026-11-0413)

## Goal Description
Provide separate Download and Upload controls for every child table in both Design DocTypes:
- **Design Contract**: `items` (`Design Contract Item`), `taxes` (`Purchase Taxes and Charges`)
- **Design Invoice**: `items` (`Design Invoice Item`), `advances` (`Design Invoice Advance`), `taxes` (`Purchase Taxes and Charges`)

The controls enable `.xlsx` round-tripping using selectable templates from `Construction Export Template`:
- Download generates table-specific `.xlsx` files with current rows or a blank template, always preserving the row identifier (`name`).
- Upload matches existing rows within the document by `name` without recreating them, creates permitted new rows, retains omitted rows, preserves advance source identity across calculations, validates types and boundaries, and rejects invalid templates, foreign identifiers, or prohibited edits atomically with row/column guidance leaving the form unchanged.

---

## User Review Required

> [!IMPORTANT]
> **Atomic Upload Rejection**: Any validation error (invalid template, duplicate/foreign identifier, prohibited modification, negative/invalid numbers) rejects the entire upload before any form state is touched, leaving the draft completely unchanged.

> [!NOTE]
> **Child Table Scope**: In line with `/ponytail ultra` minimalism, `payment_terms` (`Design Contract Term`) on `Design Contract` is excluded from Excel controls because it is hidden in the standard form layout (`hidden: 1`) and not requested in the User Story specification table.

## Proposed Changes

### Client-Side Shared Spreadsheet Module (`fabrica_construction`)

#### [MODIFY] `fabrica_construction/public/js/design_sheets.js`
Refactor and deepen `design_sheets.js` to use a declarative, unified table specification architecture. Instead of bespoke duplicated logic, define table rules (`editable_fields`, `protected_fields`, `allow_new_rows`, `required_for_new`, `numeric_fields`, `row_validator`, `post_apply`) and run them through a single atomic validation-and-apply engine.

```javascript
/** Set up Design child-table spreadsheets and apply validated inputs without replacing row identities. */

frappe.provide("fabrica_construction.design_sheets");

fabrica_construction.design_sheets.setup = function (frm, recalculate) {
    const configs = get_table_configs(frm, recalculate);
    const tables = Object.keys(configs);

    for (const fieldname of tables) {
        if (!frm.fields_dict[fieldname]?.grid) continue;
        setup_child_table(frm, fieldname, configs[fieldname], recalculate);
    }
};

function get_table_configs(frm, recalculate) {
    const is_invoice = frm.doc.doctype === "Design Invoice";
    if (is_invoice) {
        return {
            items: {
                doctype: "Design Invoice Item",
                allow_new_rows: false,
                required_export_fields: ["name", "design_contract_item"],
                editable_fields: ["qty", "accumulative_percentage"],
                protected_fields: ["design_contract_item", "rate", "contract_amount", "previous_progress", "previous_amount"],
                numeric_fields: ["qty", "accumulative_percentage"],
                validate_row(row, existing_row, row_num) {
                    if (row.design_contract_item && row.design_contract_item !== existing_row.design_contract_item) {
                        frappe.throw(__("Row {0}: Changing contract item links is prohibited.", [row_num]));
                    }
                    const qty = row.qty !== undefined && row.qty !== "" ? Number(row.qty) : existing_row.qty;
                    const progress = row.accumulative_percentage !== undefined && row.accumulative_percentage !== ""
                        ? Number(row.accumulative_percentage)
                        : existing_row.accumulative_percentage;
                    const prev_progress = Number(existing_row.previous_progress || 0);

                    if (qty < 0 || progress < prev_progress || progress > 100) {
                        frappe.throw(__("Row {0}: Quantity must be nonnegative and progress must be between previous progress and 100%.", [row_num]));
                    }
                },
            },
            advances: {
                doctype: "Design Invoice Advance",
                allow_new_rows: false,
                required_export_fields: ["name", "reference_type", "reference_name", "reference_row"],
                editable_fields: ["allocation_percentage", "allocated_amount"],
                protected_fields: [
                    "reference_type", "reference_name", "reference_row",
                    "advance_amount", "allocated_before", "remaining_balance",
                    "ref_exchange_rate", "party_exchange_rate"
                ],
                numeric_fields: ["allocation_percentage", "allocated_amount"],
                validate_row(row, existing_row, row_num) {
                    for (const ref_field of ["reference_type", "reference_name", "reference_row"]) {
                        if (row[ref_field] && String(row[ref_field]) !== String(existing_row[ref_field])) {
                            frappe.throw(__("Row {0}: Modifying advance source reference is prohibited.", [row_num]));
                        }
                    }
                    if (row.allocation_percentage !== undefined && row.allocation_percentage !== "") {
                        const pct = Number(row.allocation_percentage);
                        if (pct < 0) {
                            frappe.throw(__("Row {0}: Advance deduction cannot be negative.", [row_num]));
                        }
                        const test_amount = existing_row.advance_amount ? flt(existing_row.advance_amount) * pct / 100 : 0;
                        if (test_amount > flt(existing_row.remaining_balance)) {
                            frappe.throw(__("Row {0}: Advance deduction exceeds its remaining source balance.", [row_num]));
                        }
                    } else if (row.allocated_amount !== undefined && row.allocated_amount !== "") {
                        const amt = Number(row.allocated_amount);
                        if (amt < 0) {
                            frappe.throw(__("Row {0}: Advance deduction cannot be negative.", [row_num]));
                        }
                        if (amt > flt(existing_row.remaining_balance)) {
                            frappe.throw(__("Row {0}: Advance deduction exceeds its remaining source balance.", [row_num]));
                        }
                    }
                },
                apply_row(existing_row, values) {
                    if (values.allocation_percentage !== undefined && values.allocation_percentage !== "") {
                        existing_row.allocation_percentage = flt(values.allocation_percentage);
                    } else if (values.allocated_amount !== undefined && values.allocated_amount !== "") {
                        existing_row.allocation_percentage = flt(existing_row.advance_amount)
                            ? flt(values.allocated_amount) / flt(existing_row.advance_amount) * 100
                            : 0;
                    }
                    recalculate_advance_row(existing_row);
                },
            },
            taxes: get_taxes_config(frm),
        };
    }

    return {
        items: {
            doctype: "Design Contract Item",
            allow_new_rows: true,
            required_export_fields: ["name"],
            editable_fields: ["item_name", "description", "qty", "rate", "account", "cost_center"],
            required_new_fields: ["item_name", "qty", "rate"],
            numeric_fields: ["qty", "rate"],
            validate_row(row, existing_row, row_num) {
                const qty = row.qty !== undefined && row.qty !== "" ? Number(row.qty) : existing_row?.qty;
                const rate = row.rate !== undefined && row.rate !== "" ? Number(row.rate) : existing_row?.rate;
                if (qty !== undefined && qty < 0) {
                    frappe.throw(__("Row {0}: Quantity cannot be negative.", [row_num]));
                }
                if (rate !== undefined && rate < 0) {
                    frappe.throw(__("Row {0}: Rate cannot be negative.", [row_num]));
                }
            },
            post_apply(row) {
                row.project = frm.doc.project;
            },
        },
        taxes: get_taxes_config(frm),
    };
}

function get_taxes_config(frm) {
    return {
        doctype: "Purchase Taxes and Charges",
        allow_new_rows: true,
        required_export_fields: ["name"],
        editable_fields: [
            "charge_type", "account_head", "description", "rate",
            "tax_amount", "cost_center", "category", "add_deduct_tax", "row_id"
        ],
        required_new_fields: ["account_head"],
        numeric_fields: ["rate", "tax_amount"],
        default_new_values: {
            charge_type: "On Net Total",
            category: "Total",
            add_deduct_tax: "Add",
        },
        post_apply(row) {
            if (!row.description) {
                row.description = row.account_head;
            }
            row.project = frm.doc.project;
        },
    };
}

function recalculate_advance_row(row) {
    const default_percentage = row.advance_amount ? flt(
        flt(row.computed_default_amount) / flt(row.advance_amount) * 100,
        precision("allocation_percentage", row)
    ) : 0;
    row.allocated_amount = flt(
        row.computed_default_amount != null && flt(row.allocation_percentage) === default_percentage
            ? row.computed_default_amount
            : flt(row.advance_amount) * flt(row.allocation_percentage) / 100,
        precision("allocated_amount", row)
    );
}

function setup_child_table(frm, fieldname, config, recalculate) {
    const grid = frm.fields_dict[fieldname].grid;
    const sheets = fabrica_construction.sheet_utils;

    sheets.override_download_button(
        frm, fieldname, config.doctype, true, null,
        async (_frm, rows) => rows,
        { required_fields: config.required_export_fields, allow_blank: true }
    );

    sheets.override_upload_button(
        frm, fieldname, config.doctype,
        undefined, null, false, undefined, false, false, undefined, false, "name", false, {
            read_sheet: sheets.read_upload_sheet,
            process_upload(rows, row_numbers, fields) {
                validate_and_process_upload(frm, fieldname, config, rows, row_numbers, fields, recalculate);
            },
        }
    );

    toggle_table_buttons(frm, fieldname);
}

function validate_and_process_upload(frm, fieldname, config, rows, row_numbers, fields, recalculate) {
    const grid = frm.fields_dict[fieldname].grid;
    if (frm.doc.docstatus !== 0 || !grid.is_editable()) {
        frappe.throw(__("Only editable drafts can import data."));
    }

    // 1. Template validation
    const known_fields = new Set([
        ...config.editable_fields,
        ...(config.protected_fields || []),
        ...config.required_export_fields,
        "name"
    ]);
    const valid_columns = (fields || []).filter(f => known_fields.has(f));
    if (!fields || !fields.includes("name") || valid_columns.length <= 1) {
        frappe.throw(__("Invalid template: The uploaded sheet must include the row identifier (name) and valid table columns."));
    }

    const existing_rows = frm.doc[fieldname] || [];
    const existing_map = new Map(existing_rows.map(row => [row.name, row]));
    const names_in_sheet = new Set();

    // 2. Pure validation pass (form remains untouched on any error)
    const updates = rows.map((row, index) => {
        const row_num = row_numbers ? row_numbers[index] : index + 2;

        if (row.name) {
            if (names_in_sheet.has(row.name)) {
                frappe.throw(__("Row {0}: Duplicate row identifier '{1}'.", [row_num, row.name]));
            }
            if (!existing_map.has(row.name)) {
                frappe.throw(__("Row {0}: Identifier '{1}' does not belong to this document.", [row_num, row.name]));
            }
            names_in_sheet.add(row.name);
        } else if (!config.allow_new_rows) {
            frappe.throw(__("Row {0}: New rows are not permitted in this table. Use existing row names.", [row_num]));
        }

        const values = {};
        for (const field of config.editable_fields) {
            if (!Object.prototype.hasOwnProperty.call(row, field)) continue;
            const val = row[field];
            if (config.numeric_fields?.includes(field)) {
                if (val !== "" && val !== null && val !== undefined && !Number.isFinite(Number(val))) {
                    frappe.throw(__("Row {0}: {1} must be a number.", [row_num, field]));
                }
            }
            values[field] = val;
        }

        if (!row.name && config.required_new_fields) {
            for (const req_field of config.required_new_fields) {
                if (values[req_field] === undefined || values[req_field] === null || values[req_field] === "") {
                    frappe.throw(__("Row {0}: {1} is required for new rows.", [row_num, req_field]));
                }
            }
        }

        const existing_row = row.name ? existing_map.get(row.name) : null;
        if (config.validate_row) {
            config.validate_row(row, existing_row, row_num);
        }

        return { name: row.name, values };
    });

    // 3. Application pass (all rows passed validation)
    for (const { name, values } of updates) {
        if (name) {
            const existing_row = existing_map.get(name);
            if (config.apply_row) {
                config.apply_row(existing_row, values);
            } else {
                Object.assign(existing_row, values);
            }
            if (config.post_apply) {
                config.post_apply(existing_row);
            }
        } else {
            const new_values = { ...(config.default_new_values || {}), ...values };
            const new_row = frm.add_child(fieldname, new_values);
            if (config.post_apply) {
                config.post_apply(new_row);
            }
        }
    }

    recalculate(frm);
    frm.refresh_field(fieldname);
    frm.dirty();
}

function toggle_table_buttons(frm, fieldname) {
    const grid = frm.fields_dict[fieldname]?.grid;
    if (!grid) return;
    const can_download = [0, 1].includes(frm.doc.docstatus);
    const can_upload = frm.doc.docstatus === 0 && grid.is_editable();
    const wrapper = $(grid.wrapper);
    wrapper.find(`.grid-download-${fieldname}`).toggleClass("hidden", !can_download).toggle(can_download);
    wrapper.find(`.grid-upload-${fieldname}`).toggleClass("hidden", !can_upload).toggle(can_upload);
    wrapper.find(".grid-bulk-actions").toggle(can_download || can_upload);
}
```
